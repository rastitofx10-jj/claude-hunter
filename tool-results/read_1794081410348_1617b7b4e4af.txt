     1→"use client";
     2→
     3→// ============================================================
     4→// חדר המבצעים (Mission Control) — הלולאה השישית: ביצוע ← לקחים ← אבולוציה
     5→// מטרה בשפה טבעית ← אחזור DNA אוטומטי ← תוכנית קרב מוכנה לביצוע
     6→// דיווח שדה ← זיקוק לקחים ← DNA חדש נולד — הספרייה מחכימה מכל משימה
     7→// ============================================================
     8→
     9→import { useMemo, useState } from "react";
    10→import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
    11→import {
    12→  Target,
    13→  Loader2,
    14→  Copy,
    15→  Download,
    16→  FileText,
    17→  Send,
    18→  RotateCcw,
    19→  Trash2,
    20→  Dna,
    21→  AlertTriangle,
    22→  CheckCircle2,
    23→  ListChecks,
    24→  Sparkles,
    25→  CircleDot,
    26→  Play,
    27→  Rocket,
    28→  PackageOpen,
    29→} from "lucide-react";
    30→import { Button } from "@/components/ui/button";
    31→import { Textarea } from "@/components/ui/textarea";
    32→import { Input } from "@/components/ui/input";
    33→import { Badge } from "@/components/ui/badge";
    34→import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group";
    35→import { Label } from "@/components/ui/label";
    36→import {
    37→  Dialog,
    38→  DialogContent,
    39→  DialogDescription,
    40→  DialogFooter,
    41→  DialogHeader,
    42→  DialogTitle,
    43→} from "@/components/ui/dialog";
    44→import { toast } from "@/hooks/use-toast";
    45→import { apiFetch, downloadFile } from "@/lib/api-client";
    46→import { useHunterStore } from "@/lib/hunter-store";
    47→
    48→// ---------- types ----------
    49→
    50→interface SelectedDna {
    51→  id: number;
    52→  path: string;
    53→  name: string;
    54→  kind: string;
    55→  sourceRepo: string;
    56→  reason: string;
    57→}
    58→interface MissionPhase {
    59→  title: string;
    60→  objective: string;
    61→  steps: string[];
    62→  dnaUsed: string[];
    63→  verify: string;
    64→}
    65→interface MissionPrompt {
    66→  order: number;
    67→  label: string;
    68→  paste: string;
    69→}
    70→interface CompiledPayload {
    71→  missionName: string;
    72→  oneLiner: string;
    73→  phases: MissionPhase[];
    74→  promptSequence: MissionPrompt[];
    75→  skillMd: string;
    76→  risks: string[];
    77→  successCriteria: string[];
    78→  selectedDna: SelectedDna[];
    79→}
    80→interface MissionListItem {
    81→  id: number;
    82→  missionName: string;
    83→  goal: string;
    84→  status: string;
    85→  oneLiner: string;
    86→  phaseCount: number;
    87→  promptCount: number;
    88→  hasReport: boolean;
    89→  createdAt: string;
    90→}
    91→interface MissionListResponse {
    92→  items: MissionListItem[];
    93→  stats: { totalDna: number; fieldDna: number };
    94→}
    95→interface MissionDetail extends CompiledPayload {
    96→  id: number;
    97→  goal: string;
    98→  context: string | null;
    99→  status: string;
   100→  report: { outcome: string; verdict: string; lessons: string[]; bornDnaId: number | null; reportedAt: string } | null;
   101→  autopilot: { steps: { phaseIndex: number; title: string; deliverable: string; executedAt: string }[]; failures: string[] } | null;
   102→  autopilotAt: string | null;
   103→  createdAt: string;
   104→  bundle: string;
   105→}
   106→interface ReportResponse {
   107→  ok: boolean;
   108→  verdict: string;
   109→  lessons: string[];
   110→  bornDna: { id: number | null; name: string; description: string } | null;
   111→}
   112→
   113→// ---------- constants ----------
   114→
   115→const EXAMPLE_GOALS = [
   116→  "לבנות סקיל של סינתזת חיפוש-רשת שמפיק סקירה מאומתת עם מקורות",
   117→  "לשדרג את תהליך הבדיקות שלנו ל-E2E אוטונומי בדפדפן",
   118→  "לתכנן ארכיטקטורה למודול AI חדש: מטמון, ריטריים וזרימת סטרימינג",
   119→];
   120→
   121→const STATUS_META: Record<string, { label: string; cls: string }> = {
   122→  compiled: { label: "הודרה", cls: "bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-300" },
   123→  in_progress: { label: "בביצוע", cls: "bg-violet-100 text-violet-800 dark:bg-violet-950 dark:text-violet-300" },
   124→  done: { label: "הושלמה", cls: "bg-emerald-100 text-emerald-800 dark:bg-emerald-950 dark:text-emerald-300" },
   125→  partial: { label: "חלקית", cls: "bg-orange-100 text-orange-800 dark:bg-orange-950 dark:text-orange-300" },
   126→  failed: { label: "נכשלה", cls: "bg-rose-100 text-rose-800 dark:bg-rose-950 dark:text-rose-300" },
   127→};
   128→
   129→const OUTCOME_OPTIONS: { value: string; label: string; hint: string }[] = [
   130→  { value: "success", label: "הצלחה מלאה", hint: "כל קריטריוני ההצלחה עברו" },
   131→  { value: "partial", label: "הצלחה חלקית", hint: "חלק מהשלבים עברו, יש מה לתקן" },
   132→  { value: "failed", label: "נכשלה", hint: "נתקענו — אבל יש לקחים יקרים" },
   133→];
   134→
   135→// ---------- component ----------
   136→
   137→export function MissionControlPanel() {
   138→  const token = useHunterStore((s) => s.token);
   139→  const qc = useQueryClient();
   140→  const [goal, setGoal] = useState("");
   141→  const [context, setContext] = useState("");
   142→  const [detail, setDetail] = useState<MissionDetail | null>(null);
   143→  const [reportOpen, setReportOpen] = useState(false);
   144→  const [outcome, setOutcome] = useState("success");
   145→  const [notes, setNotes] = useState("");
   146→
   147→  // ---------- queries ----------
   148→  const { data: listData, isLoading: listLoading } = useQuery({
   149→    queryKey: ["missions"],
   150→    queryFn: () => apiFetch<MissionListResponse>("/api/mission", token),
   151→  });
   152→
   153→  // ---------- mutations ----------
   154→  const compileM = useMutation({
   155→    mutationFn: () =>
   156→      apiFetch<{ id: number } & CompiledPayload>("/api/mission", token, {
   157→        method: "POST",
   158→        body: JSON.stringify({ goal: goal.trim(), context: context.trim() }),
   159→      }),
   160→    onSuccess: (data) => {
   161→      setDetail({ ...data, goal: goal.trim(), context: context.trim() || null, status: "compiled", report: null, createdAt: new Date().toISOString() });
   162→      void qc.invalidateQueries({ queryKey: ["missions"] });
   163→      toast({ title: "משימה הודרה 🎯", description: `${data.missionName} · ${data.phases.length} שלבים · נבחרו ${data.selectedDna.length} גנים` });
   164→    },
   165→    onError: (e: Error) => toast({ title: "ההידור נכשל", description: e.message, variant: "destructive" }),
   166→  });
   167→
   168→  const statusM = useMutation({
   169→    mutationFn: (p: { id: number; status: string }) =>
   170→      apiFetch<{ id: number; status: string }>(`/api/mission/${p.id}`, token, {
   171→        method: "PATCH",
   172→        body: JSON.stringify({ status: p.status }),
   173→      }),
   174→    onSuccess: (data) => {
   175→      setDetail((d) => (d && d.id === data.id ? { ...d, status: data.status } : d));
   176→      void qc.invalidateQueries({ queryKey: ["missions"] });
   177→    },
   178→    onError: (e: Error) => toast({ title: "עדכון סטטוס נכשל", description: e.message, variant: "destructive" }),
   179→  });
   180→
   181→  const reportM = useMutation({
   182→    mutationFn: (p: { missionId: number }) =>
   183→      apiFetch<ReportResponse>("/api/mission/report", token, {
   184→        method: "POST",
   185→        body: JSON.stringify({ missionId: p.missionId, outcome, notes: notes.trim() }),
   186→      }),
   187→    onSuccess: (data, vars) => {
   188→      setReportOpen(false);
   189→      setNotes("");
   190→      setDetail((d) =>
   191→        d && d.id === vars.missionId
   192→          ? { ...d, status: outcome === "success" ? "done" : outcome, report: { outcome, verdict: data.verdict, lessons: data.lessons, bornDnaId: data.bornDna?.id ?? null, reportedAt: new Date().toISOString() } }
   193→          : d
   194→      );
   195→      void qc.invalidateQueries({ queryKey: ["missions"] });
   196→      void qc.invalidateQueries({ queryKey: ["dna-library"] });
   197→      toast({
   198→        title: data.bornDna ? "לקחים זוקקו · DNA חדש נולד 🧬" : "לקחים זוקקו ✅",
   199→        description: data.bornDna ? `נולד: ${data.bornDna.name} — הספרייה מחכימה` : data.verdict,
   200→      });
   201→    },
   202→    onError: (e: Error) => toast({ title: "זיקוק הדיווח נכשל", description: e.message, variant: "destructive" }),
   203→  });
   204→
   205→  // ---------- טייס אוטומטי ----------
   206→  const autopilotM = useMutation({
   207→    mutationFn: (id: number) =>
   208→      apiFetch<{ ok: boolean; steps: { phaseIndex: number; title: string; deliverable: string; executedAt: string }[]; failures: string[]; totalPhases: number }>(
   209→        `/api/mission/${id}/autopilot`,
   210→        token,
   211→        { method: "POST" }
   212→      ),
   213→    onSuccess: (data, id) => {
   214→      setDetail((d) => (d && d.id === id ? { ...d, status: d.status === "compiled" ? "in_progress" : d.status, autopilot: { steps: data.steps, failures: data.failures }, autopilotAt: new Date().toISOString() } : d));
   215→      void qc.invalidateQueries({ queryKey: ["missions"] });
   216→      toast({
   217→        title: `הטייס האוטומטי נחת 🚀`,
   218→        description: `בוצעו ${data.steps.length}/${data.totalPhases} שלבים · ${data.failures.length > 0 ? `${data.failures.length} כשלים` : "בלי כשלים"}`,
   219→      });
   220→    },
   221→    onError: (e: Error) => toast({ title: "הריצה האוטונומית נכשלה", description: e.message, variant: "destructive" }),
   222→  });
   223→
   224→  const deleteM = useMutation({
   225→    mutationFn: (id: number) => apiFetch<{ ok: boolean }>(`/api/mission/${id}`, token, { method: "DELETE" }),
   226→    onSuccess: (_d, id) => {
   227→      setDetail((d) => (d?.id === id ? null : d));
   228→      void qc.invalidateQueries({ queryKey: ["missions"] });
   229→      toast({ title: "המשימה נמחקה" });
   230→    },
   231→    onError: (e: Error) => toast({ title: "המחיקה נכשלה", description: e.message, variant: "destructive" }),
   232→  });
   233→
   234→  const loadDetail = async (id: number) => {
   235→    try {
   236→      const d = await apiFetch<MissionDetail>(`/api/mission/${id}`, token);
   237→      setDetail(d);
   238→    } catch (e) {
   239→      toast({ title: "טעינת המשימה נכשלה", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   240→    }
   241→  };
   242→
   243→  const copyText = async (text: string, what: string) => {
   244→    try {
   245→      await navigator.clipboard.writeText(text);
   246→      toast({ title: `${what} הועתק 📋` });
   247→    } catch {
   248→      toast({ title: "ההעתקה נכשלה", variant: "destructive" });
   249→    }
   250→  };
   251→
   252→  const stats = listData?.stats;
   253→  const items = useMemo(() => listData?.items ?? [], [listData]);
   254→
   255→  return (
   256→    <div className="space-y-5">
   257→      {/* לולאת האבולוציה — מחוון סטטוס */}
   258→      <section className="rounded-xl border bg-gradient-to-l from-violet-50/70 via-background to-background p-4 dark:from-violet-950/20" aria-label="מחוון לולאת האבולוציה">
   259→        <div className="flex flex-wrap items-center justify-between gap-3">
   260→          <div className="flex items-center gap-2">
   261→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-violet-600 text-white shadow-sm">
   262→              <RotateCcw className="h-4.5 w-4.5" aria-hidden />
   263→            </div>
   264→            <div>
   265→              <p className="text-sm font-bold">לולאת האבולוציה</p>
   266→              <p className="text-xs text-muted-foreground">בצע משימה ← דווח מהשדה ← לקחים הופכים ל-DNA חדש ← הספרייה מחכימה</p>
   267→            </div>
   268→          </div>
   269→          <div className="flex items-center gap-2 text-xs">
   270→            <Badge variant="outline" className="gap-1 px-2.5 py-1">
   271→              <Dna className="h-3.5 w-3.5" /> {stats?.totalDna ?? 0} גנים בספרייה
   272→            </Badge>
   273→            <Badge className="gap-1 border-emerald-300 bg-emerald-50 px-2.5 py-1 text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950/60 dark:text-emerald-300">
   274→              <Sparkles className="h-3.5 w-3.5" /> {stats?.fieldDna ?? 0} נולדו מהשדה
   275→            </Badge>
   276→          </div>
   277→        </div>
   278→      </section>
   279→
   280→      <div className="grid gap-5 xl:grid-cols-3">
   281→        {/* עמודת ההידור והפרטים */}
   282→        <div className="space-y-5 xl:col-span-2">
   283→          {/* טופס הידור */}
   284→          <section className="rounded-xl border bg-card p-4 shadow-sm" aria-label="הידור משימה חדשה">
   285→            <p className="mb-1 flex items-center gap-2 text-sm font-bold">
   286→              <Target className="h-4 w-4 text-violet-600" /> הדבר משימה חדשה
   287→            </p>
   288→            <p className="mb-3 text-xs text-muted-foreground">
   289→              תאר מטרה בשפה טבעית — המנוע יבחר לבד את ה-DNA הרלוונטי מהספרייה וירכיב תוכנית קרב + רצף פרומפטים מוכן להדבקה ב-Claude Code
   290→            </p>
   291→            <Textarea
   292→              value={goal}
   293→              onChange={(e) => setGoal(e.target.value)}
   294→              placeholder="לדוגמה: לבנות סקיל שמבצע חיפוש רשת, מצליב מקורות ומפיק דוח מאומת…"
   295→              className="mb-2 min-h-20 resize-y"
   296→              aria-label="מטרת המשימה"
   297→            />
   298→            <Input
   299→              value={context}
   300→              onChange={(e) => setContext(e.target.value)}
   301→              placeholder="הקשר נוסף (רשות): סטאק, אילוצים, דדליין…"
   302→              className="mb-3 text-sm"
   303→              aria-label="הקשר נוסף"
   304→            />
   305→            <div className="mb-3 flex flex-wrap gap-1.5">
   306→              {EXAMPLE_GOALS.map((g) => (
   307→                <button
   308→                  key={g}
   309→                  onClick={() => setGoal(g)}
   310→                  className="rounded-full border px-2.5 py-1 text-[11px] text-muted-foreground transition-colors hover:border-violet-400 hover:text-violet-700 dark:hover:text-violet-300"
   311→                >
   312→                  {g}
   313→                </button>
   314→              ))}
   315→            </div>
   316→            <Button
   317→              onClick={() => compileM.mutate()}
   318→              disabled={compileM.isPending || goal.trim().length < 8}
   319→              className="w-full bg-violet-600 text-white hover:bg-violet-700 sm:w-auto"
   320→            >
   321→              {compileM.isPending ? (
   322→                <>
   323→                  <Loader2 className="ml-2 h-4 w-4 animate-spin" /> מרכיב תוכנית קרב… (עד דקה)
   324→                </>
   325→              ) : (
   326→                <>
   327→                  <Play className="ml-2 h-4 w-4" /> הדר משימה
   328→                </>
   329→              )}
   330→            </Button>
   331→          </section>
   332→
   333→          {/* משימה פעילה — פירוט */}
   334→          {detail && (
   335→            <MissionDetailCard
   336→              m={detail}
   337→              onCopy={copyText}
   338→              onStatus={(s) => statusM.mutate({ id: detail.id, status: s })}
   339→              onReport={() => setReportOpen(true)}
   340→              onDelete={() => deleteM.mutate(detail.id)}
   341→              onAutopilot={() => autopilotM.mutate(detail.id)}
   342→              busy={statusM.isPending || deleteM.isPending || autopilotM.isPending}
   343→              autopilotRunning={autopilotM.isPending}
   344→            />
   345→          )}
   346→        </div>
   347→
   348→        {/* עמודת ההיסטוריה */}
   349→        <section className="rounded-xl border bg-card p-4 shadow-sm" aria-label="משימות קודמות">
   350→          <p className="mb-3 flex items-center gap-2 text-sm font-bold">
   351→            <ListChecks className="h-4 w-4 text-violet-600" /> משימות ({items.length})
   352→          </p>
   353→          {listLoading ? (
   354→            <div className="space-y-2">
   355→              {Array.from({ length: 3 }).map((_, i) => (
   356→                <div key={i} className="h-16 animate-pulse rounded-lg bg-muted/50" />
   357→              ))}
   358→            </div>
   359→          ) : items.length === 0 ? (
   360→            <div className="rounded-lg border border-dashed p-6 text-center text-xs text-muted-foreground">
   361→              עדיין אין משימות. הדבר מטרה והתחל לבצע 🚀
   362→            </div>
   363→          ) : (
   364→            <ul className="max-h-[560px] space-y-2 overflow-y-auto pr-1">
   365→              {items.map((it) => {
   366→                const st = STATUS_META[it.status] ?? STATUS_META.compiled;
   367→                return (
   368→                  <li key={it.id}>
   369→                    <button
   370→                      onClick={() => void loadDetail(it.id)}
   371→                      className={`w-full rounded-lg border p-3 text-right transition-colors hover:border-violet-400 hover:bg-violet-50/50 dark:hover:bg-violet-950/20 ${detail?.id === it.id ? "border-violet-500 bg-violet-50/60 dark:bg-violet-950/30" : ""}`}
   372→                    >
   373→                      <div className="flex items-center justify-between gap-2">
   374→                        <span dir="ltr" className="truncate font-mono text-xs font-semibold text-violet-700 dark:text-violet-300">
   375→                          {it.missionName}
   376→                        </span>
   377→                        <span className={`shrink-0 rounded-full px-2 py-0.5 text-[10px] font-bold ${st.cls}`}>{st.label}</span>
   378→                      </div>
   379→                      <p className="mt-1 line-clamp-2 text-xs text-muted-foreground">{it.goal}</p>
   380→                      <div className="mt-1.5 flex flex-wrap items-center gap-1 text-[10px] text-muted-foreground">
   381→                        <span>{it.phaseCount} שלבים</span>
   382→                        <span>·</span>
   383→                        <span>{it.promptCount} פרומפטים</span>
   384→                        {it.hasReport && (
   385→                          <>
   386→                            <span>·</span>
   387→                            <span className="font-semibold text-emerald-600 dark:text-emerald-400">דווח ✓</span>
   388→                          </>
   389→                        )}
   390→                      </div>
   391→                    </button>
   392→                  </li>
   393→                );
   394→              })}
   395→            </ul>
   396→          )}
   397→        </section>
   398→      </div>
   399→
   400→      {/* דיאלוג דיווח שדה */}
   401→      <Dialog open={reportOpen} onOpenChange={setReportOpen}>
   402→        <DialogContent className="sm:max-w-lg">
   403→          <DialogHeader>
   404→            <DialogTitle className="flex items-center gap-2">
   405→              <Send className="h-4 w-4 text-violet-600" /> דיווח שדה — הזן את הלולאה
   406→            </DialogTitle>
   407→            <DialogDescription>
   408→              ספר מה קרה בפועל. המזקק יפיק לקחים ויוליד DNA חדש בספרייה — כך המערכת משתפרת מכל משימה.
   409→            </DialogDescription>
   410→          </DialogHeader>
   411→
   412→          <RadioGroup value={outcome} onValueChange={setOutcome} className="my-2 grid gap-2">
   413→            {OUTCOME_OPTIONS.map((o) => (
   414→              <Label
   415→                key={o.value}
   416→                htmlFor={`outcome-${o.value}`}
   417→                className={`flex cursor-pointer items-start gap-3 rounded-lg border p-3 transition-colors ${outcome === o.value ? "border-violet-500 bg-violet-50/60 dark:bg-violet-950/30" : "hover:bg-muted/40"}`}
   418→              >
   419→                <RadioGroupItem id={`outcome-${o.value}`} value={o.value} className="mt-0.5" />
   420→                <span>
   421→                  <span className="block text-sm font-semibold">{o.label}</span>
   422→                  <span className="block text-xs text-muted-foreground">{o.hint}</span>
   423→                </span>
   424→              </Label>
   425→            ))}
   426→          </RadioGroup>
   427→
   428→          <Textarea
   429→            value={notes}
   430→            onChange={(e) => setNotes(e.target.value)}
